  var reader = $('reader');
  var readerContent = $('readerContent');
  var readerBody = $('readerBody');
  var readerTitleBar = $('readerTitleBar');

  function openArticle(id){
    var a = null;
    for (var i = 0; i < ARTICLES.length; i++){
      if (ARTICLES[i].id === id){ a = ARTICLES[i]; break; }
    }
    if (!a) return;
    readerContent.innerHTML = a.content;
    readerTitleBar.textContent = a.t;
    reader.classList.add('on');
    document.body.style.overflow = 'hidden';
    readerBody.scrollTop = 0;
  }
  function closeArticle(){
    reader.classList.remove('on');
    document.body.style.overflow = '';
  }
  $('readerBack').addEventListener('click', closeArticle);
  document.addEventListener('keydown', function(e){
    if (e.key === 'Escape' && reader.classList.contains('on')) closeArticle();
  });
  readerBody.addEventListener('scroll', function(){
    var max = readerBody.scrollHeight - readerBody.clientHeight;
    var pct = max > 0 ? (readerBody.scrollTop / max) * 100 : 0;
    $('readerProgress').style.width = pct + '%';
  }, { passive: true });

  /* REVEAL */
  var io = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){ e.target.classList.add('in'); io.unobserve(e.target); }
    });
  }, { threshold: 0.08, rootMargin: '0px 0px -40px 0px' });
  function observeRv(){
    var els = document.querySelectorAll('.rv:not(.in)');
    for (var i = 0; i < els.length; i++) io.observe(els[i]);
  }

  /* COUNTERS */
  var countersDone = false;
  function animateCounters(){
    if (countersDone) return;
    countersDone = true;
    var els = document.querySelectorAll('[data-count]');
    for (var i = 0; i < els.length; i++){
      (function(el){
        var target = parseInt(el.getAttribute('data-count'), 10);
        var n = 0;
        var step = Math.max(1, Math.round(target / 60));
        var timer = setInterval(function(){
          n += step;
          if (n >= target){ n = target; clearInterval(timer); }
          el.textContent = n + (target >= 1000 ? '+' : '');
        }, 25);
      })(els[i]);
    }
  }
  var statsIO = new IntersectionObserver(function(entries){
    entries.forEach(function(e){
      if (e.isIntersecting){ animateCounters(); statsIO.disconnect(); }
    });
  }, { threshold: 0.5 });
  statsIO.observe(document.querySelector('.hero-stats'));
